example (p : Usize) (hp : p.val < 32) (alpha : Array I64 32#usize)
    (vs : Array (Array I64 32#usize) 32#usize)
    (ha : ∀ j (_ : j ≤ p.val), 0 ≤ (ints alpha)[j]'(by rw [ints_length32]; omega))
    (hs : ((ints alpha).take (p.val + 1)).sum ≤ 16777216)
    (hv : ∀ row ∈ vs.val, Bounded Envelope.value row) :
    general_engine.head_out_at_loop1 p alpha vs (Array.repeat 32#usize 0#i64) 0#usize ⦃ r =>
      ints r = GeneralModel.mixAt (ints alpha) (vals2 vs) p.val ∧ Bounded Envelope.mixed_value r ⦄ :=
  mix_scan_spec p hp alpha vs ha hs hv
