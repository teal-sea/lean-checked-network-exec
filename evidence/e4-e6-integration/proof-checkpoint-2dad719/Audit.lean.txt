open LcnExec.GeneralEngine
example (W : general_engine.Weights) (hW : LoadBounds W)
    (l : Usize) (hl : l.val < 2)
    (r : Array I64 128#usize) (hr : Bounded Envelope.residual r) :
    general_engine.mlp_hidden W l r ⦃ m =>
      ints m = GeneralModel.mlpHidden (toGeneralModel W) l.val (ints r) ∧
      Bounded Envelope.hidden m ⦄ :=
  mlp_hidden_spec W hW l hl r hr
