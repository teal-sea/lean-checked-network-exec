import LcnExec.GeneralEngine.HeadOut

/-!
# The four-head attention accumulator

`attention_at_loop0_loop0` adds a 128-entry head output into the current accumulator;
`attention_at_loop0` folds the four heads; `attention_at` is the whole attention block.
-/

open Aeneas Aeneas Std Result ControlFlow

set_option maxHeartbeats 800000

namespace LcnExec.GeneralEngine

/-- Inner pointwise checked add: `att[k] := att[k] + ho[k]` for all `k < 128`. -/
theorem att_add_spec (att ho : Array I64 128#usize) (b : Int)
    (hatt : Bounded b att) (hho : Bounded Envelope.head_output ho)
    (hb : b ≤ 3 * Envelope.head_output) :
    general_engine.attention_at_loop0_loop0 att ho 0#usize ⦃ r =>
      ints r = GeneralModel.addV (ints att) (ints ho) ∧
      Bounded (b + Envelope.head_output) r ⦄ := by
  have hhead : 0 ≤ Envelope.head_output := by
    simp only [Envelope.head_output]; norm_num
  unfold general_engine.attention_at_loop0_loop0
  apply loop.spec_decr_nat
    (measure := fun (st : Array I64 128#usize × Usize) => 128 - st.2.val)
    (inv := fun (st : Array I64 128#usize × Usize) =>
      st.2.val ≤ 128 ∧
      Bounded (b + Envelope.head_output) st.1 ∧
      (∀ (k : Nat) (hk : k < 128) (hik : k < st.2.val),
        (ints st.1)[k]'(by rw [ints_length128]; omega) =
          (ints att)[k]'(by rw [ints_length128]; omega) +
            (ints ho)[k]'(by rw [ints_length128]; omega)))
  · rintro ⟨att1, k⟩ ⟨hk1, hbd, heq⟩
    unfold general_engine.attention_at_loop0_loop0.body
    dsimp only at hk1 hbd heq ⊢
    simp only [general_engine.D_MODEL]
    split
    · have hk128 : k.val < 128 := by scalar_tac
      step with Array.index_usize_spec ho k (by simp; omega) as ⟨i, hiv⟩
      step with Array.index_usize_spec att1 k (by simp; omega) as ⟨i1, hi1v⟩
      have hminb : I64.min ≤ i1.val + i.val := by scalar_tac
      have hmaxb : i1.val + i.val ≤ I64.max := by scalar_tac
      step with I64.add_spec hminb hmaxb as ⟨i2, hi2⟩
      step with Array.update_spec att1 k i2 (by simp; omega) as ⟨a, ha⟩
      step as ⟨k1, hk1'⟩
      simp only [hk1']
      refine ⟨by omega, ?_, ?_, by omega⟩
      · intro v hv
        have hvset : v ∈ att1.val.set k.val i2 := by
          rw [ha]; exact Array.set_mem??PLACEHOLDER v hv
        rcases List.mem_map.mp (by simpa [ints] using hvset) with ⟨r, hr, rfl⟩
        by_cases hkk : (r : Nat) = k.val
        · have hself : (att1.val.set k.val i2)[k.val]'(by simp; omega) = i2 :=
            List.getElem_set_self (by simp; omega)
          have hval : (ints a)[k.val]'(by rw [ints_length128]; omega) = i2.val := by
            simp only [ints, List.getElem_map, ha, Array.set_val_eq]
            rw [hself]
          rw [hval, hi2]
          constructor <;> omega
        · have hne : r.val ≠ k.val := by omega
          have hne2 : (att1.val.set k.val i2)[r.val]??PLACEHOLDER2
          sorry
      · intro k' hk' hik
        by_cases hkk : k' = k.val
        · subst k'
          have hval : (ints a)[k.val]'(by rw [ints_length128]; omega) =
              i1.val + i.val := by
            rw [ha, ints, Array.set_val_eq, List.getElem_map]
            rw [List.getElem_set_self (by simp; omega)]
            sorry
          have hsrc : (ints att1)[k.val]'(by rw [ints_length128]; omega) = i1.val := by
            simp only [ints, List.getElem_map]
            exact hi1v
          have hho : (ints ho)[k.val]'(by rw [ints_length128]; omega) = i.val := by
            simp only [ints, List.getElem_map]
            exact hiv
          rw [hval, hsrc, hho] have hval : (ints a)[k']'(by rw [ints_length128]; omega) =
              (ints att1)[k']'(by rw [ints_length128]; omega) := by
            simp only [ha, ints, Array.set_val_eq, List.getElem_map]
            rw [List.getElem_set_ne (by omega)]
          rw [hval]
          exact heq k' hk' (by omega)
    · have hk128 : k.val = 128 := by scalar_tac
      simp only [WP.spec_ok]
      refine ⟨?_, hbd⟩
      apply List.ext_getElem
      · simp [ints_length128, project_addV_length]
      · intro k' hk' hk''
        rw [project_addV_get _ _ (by simp [ints_length128]) k'
          (by rw [ints_length128]; omega)]
        exact heq k' (by rw [ints_length128] at hk'; omega) (by omega)
  · simp only
    refine ⟨by scalar_tac, ?_, ?_⟩
    · intro v hv
      have := hatt v hv
      constructor <;> omega
    · intro k' hk' hik
      exact absurd hik (by show ¬(k' < 0); omega)

end LcnExec.GeneralEngine
