  embed_all_spec W hW t n hn hv

    (n : Usize) (hp : 0 < n.val) (hn : n.val ≤ 32) (hw : W.n_layers.val ≤ 2)

        W.u W.exp_n W.exp_2 W.exp_1 W.exp_0 n x last 0#usize ⦃ out =>
      ints out = GeneralModel.runLayers (toGeneralModel W) 0 W.n_layers.val
        (ints2 x) n.val (ints last) ∧ Bounded Envelope.final_stream out ⦄ :=
  forward_logits_loop1_spec W hW n hp hn hw x hx last hlast
